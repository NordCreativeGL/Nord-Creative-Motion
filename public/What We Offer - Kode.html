class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.mountRef = React.createRef();
    this.labelRef = React.createRef();
    this.disposed = false;
  }

  renderVals() {
    return { mountRef: this.mountRef, labelRef: this.labelRef };
  }

  async componentDidMount() {
    const THREE = await this.loadThree();
    if (this.disposed || !this.mountRef.current) return;
    this.THREE = THREE;
    this.initScene();
  }

  componentWillUnmount() {
    this.disposed = true;
    if (this._raf) cancelAnimationFrame(this._raf);
    if (this.renderer) {
      if (this._onMove) { this.renderer.domElement.removeEventListener('pointermove', this._onMove); this.renderer.domElement.removeEventListener('pointerdown', this._onMove); }
      this.renderer.dispose();
    }
    if (this._onResize) window.removeEventListener('resize', this._onResize);
  }

  loadThree() {
    return new Promise((resolve) => {
      if (window.THREE) return resolve(window.THREE);
      // import the module build and stash on window
      import('https://cdnjs.cloudflare.com/ajax/libs/three.js/0.160.0/three.module.min.js')
        .then((mod) => { window.THREE = mod; resolve(mod); });
    });
  }

  initScene() {
    const THREE = this.THREE;
    const host = this.mountRef.current;
    const w = host.clientWidth || window.innerWidth;
    const h = host.clientHeight || window.innerHeight;

    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false, preserveDrawingBuffer: true });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(w, h);
    renderer.setClearColor(0x0a1c28, 1);
    host.appendChild(renderer.domElement);
    this.renderer = renderer;

    const scene = new THREE.Scene();
    scene.fog = new THREE.FogExp2(0x3a7280, 0.0023);
    this.scene = scene;

    const camera = new THREE.PerspectiveCamera(60, w / h, 0.1, 1600);
    camera.position.set(0, 16, 40);
    camera.lookAt(0, 9, -95);
    this.camera = camera;

    this.t = 0;
    this.buildLights();
    this.buildSky();
    this.buildInlandIce();
    this.buildGlacier();
    this.buildSideMountains();
    this.buildWater();

    this._onResize = () => {
      const W = host.clientWidth, H = host.clientHeight;
      renderer.setSize(W, H);
      camera.aspect = W / H;
      camera.updateProjectionMatrix();
    };
    window.addEventListener('resize', this._onResize);

    const clock = new THREE.Clock();
    window.__wwo = { frames: 0, err: null };
    window.__wwoRender = () => renderer.render(scene, camera);
    window.__wwoDebug = this;
    const loop = () => {
      if (this.disposed) return;
      this._raf = requestAnimationFrame(loop);
      try {
        const dt = Math.min(clock.getDelta(), 0.05);
        this.t += dt;
        this.updateScene(dt, this.t);
        renderer.render(scene, camera);
        window.__wwo.frames++;
      } catch (e) { window.__wwo.err = e.message; cancelAnimationFrame(this._raf); }
    };
    loop();
  }

  // ---------- noise helpers ----------
  _hash(x, y) { const s = Math.sin(x * 127.1 + y * 311.7) * 43758.5453; return s - Math.floor(s); }
  _vnoise(x, y) {
    const xi = Math.floor(x), yi = Math.floor(y), xf = x - xi, yf = y - yi;
    const u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf);
    const a = this._hash(xi, yi), b = this._hash(xi + 1, yi), c = this._hash(xi, yi + 1), d = this._hash(xi + 1, yi + 1);
    return a * (1 - u) * (1 - v) + b * u * (1 - v) + c * (1 - u) * v + d * u * v;
  }
  _fbm(x, y) { let s = 0, amp = 1, f = 1, n = 0; for (let i = 0; i < 4; i++) { s += amp * this._vnoise(x * f, y * f); n += amp; amp *= 0.5; f *= 2; } return s / n; }

  buildLights() {
    const THREE = this.THREE, s = this.scene;
    s.add(new THREE.AmbientLight(0x274a59, 1.45));
    s.add(new THREE.HemisphereLight(0x4a7a8a, 0x0c2028, 1.5));
    const moon = new THREE.DirectionalLight(0xe6f3ff, 1.9);
    moon.position.set(-34, 60, 30); s.add(moon);
    const fill = new THREE.DirectionalLight(0xbcd6e8, 1.0);
    fill.position.set(40, 44, 24); s.add(fill);
    const teal = new THREE.PointLight(0x3fc2b4, 6.0, 220, 2);
    teal.position.set(0, 2, 14); s.add(teal); this.tealLight = teal;
    const teal2 = new THREE.PointLight(0x29928a, 3.2, 150, 2);
    teal2.position.set(-22, 1.5, -6); s.add(teal2);
    const teal3 = new THREE.PointLight(0x29928a, 2.8, 150, 2);
    teal3.position.set(22, 1.5, -6); s.add(teal3);
  }

  buildSky() {
    const THREE = this.THREE;
    // gradient backdrop (navy -> teal horizon), matching the hero
    const cv = document.createElement('canvas'); cv.width = 16; cv.height = 320;
    const cx = cv.getContext('2d');
    const grd = cx.createLinearGradient(0, 0, 0, 320);
    grd.addColorStop(0, '#0a1a28');
    grd.addColorStop(0.34, '#0e2533');
    grd.addColorStop(0.5, '#1b4452');
    grd.addColorStop(0.58, '#2d6478');
    grd.addColorStop(0.66, '#1d4858');
    grd.addColorStop(1, '#0c2a34');
    cx.fillStyle = grd; cx.fillRect(0, 0, 16, 320);
    const bg = new THREE.CanvasTexture(cv); bg.colorSpace = THREE.SRGBColorSpace;
    this.scene.background = bg;
    const N = 1300, pos = new Float32Array(N * 3);
    for (let i = 0; i < N; i++) {
      const r = 380 + Math.random() * 320, th = Math.random() * Math.PI * 2;
      pos[i * 3] = Math.cos(th) * r;
      pos[i * 3 + 1] = 110 + Math.random() * 300;
      pos[i * 3 + 2] = -220 - Math.random() * 420;
    }
    const g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    const m = new THREE.PointsMaterial({ color: 0xdcecff, size: 1.4, sizeAttenuation: true, transparent: true, opacity: 0.7 });
    this.stars = new THREE.Points(g, m);
    this.scene.add(this.stars);
  }

  buildHorizon() {
    const THREE = this.THREE, M = THREE.MathUtils;
    // distant land/ice ridge across the full width — the horizon far inland behind the glacier
    const cols = 120, rows = 8, stride = cols + 1, verts = [];
    const zR = -640, spanX = 820;
    for (let i = 0; i <= rows; i++) {
      const t = i / rows;                       // 0 ridge crest .. 1 base
      for (let j = 0; j <= cols; j++) {
        const u = j / cols;
        const x = (u - 0.5) * spanX;
        const ridge = (this._fbm(u * 7, 2.3) * 0.7 + this._fbm(u * 18 + 4, 5) * 0.3);
        const crest = 14 + ridge * 30;
        const y = M.lerp(crest, -6, t);
        verts.push(x, y, zR + (this._vnoise(u * 9, t * 3) - 0.5) * 30);
      }
    }
    const idx = [];
    for (let i = 0; i < rows; i++) for (let j = 0; j < cols; j++) {
      const a = i * stride + j, b = a + 1, c = a + stride, d = c + 1;
      idx.push(a, c, b, b, c, d);
    }
    let geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
    geo.setIndex(idx);
    geo = geo.toNonIndexed();
    geo.computeVertexNormals();
    const p = geo.attributes.position, tc = p.count / 3;
    const col = new Float32Array(p.count * 3);
    const cLow = new THREE.Color(0x163a47), cHi = new THREE.Color(0x37606f), tmp = new THREE.Color();
    for (let t = 0; t < tc; t++) {
      const a = 3 * t;
      const yc = (p.getY(a) + p.getY(a + 1) + p.getY(a + 2)) / 3;
      tmp.copy(cLow).lerp(cHi, M.clamp(yc / 40, 0, 1));
      for (let k = 0; k < 3; k++) { col[(a + k) * 3] = tmp.r; col[(a + k) * 3 + 1] = tmp.g; col[(a + k) * 3 + 2] = tmp.b; }
    }
    geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
    const mat = new THREE.MeshBasicMaterial({ vertexColors: true, fog: true });
    this.scene.add(new THREE.Mesh(geo, mat));
  }

  buildInlandIce() {
    const THREE = this.THREE, M = THREE.MathUtils;
    // distant snowy inland ice cap continuing over the saddle, behind the glacier crest
    const cols = 110, rows = 16, stride = cols + 1, verts = [];
    const zR = -298, spanX = 540;
    for (let i = 0; i <= rows; i++) {
      const t = i / rows;                       // 0 front crest .. 1 far
      for (let j = 0; j <= cols; j++) {
        const u = j / cols;
        const x = (u - 0.5) * spanX;
        const swell = this._fbm(u * 5 + 2, t * 3 + 1) * 26 + this._fbm(u * 13, t * 6) * 10;
        const dip = Math.pow(Math.abs(u - 0.5) * 2, 1.6) * 6;  // gentle saddle low in the middle
        const y = 74 + swell - dip + t * 10;                    // broad ice field rising to the horizon
        verts.push(x, y, zR - t * 150 + (this._vnoise(u * 8, t * 4) - 0.5) * 24);
      }
    }
    const idx = [];
    for (let i = 0; i < rows; i++) for (let j = 0; j < cols; j++) {
      const a = i * stride + j, b = a + 1, c = a + stride, d = c + 1;
      idx.push(a, c, b, b, c, d);
    }
    let geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
    geo.setIndex(idx);
    geo = geo.toNonIndexed();
    geo.computeVertexNormals();
    const p = geo.attributes.position, nrm = geo.attributes.normal, tc = p.count / 3;
    const col = new Float32Array(p.count * 3);
    const cSnow = new THREE.Color(0xe6f3f2), cBlue = new THREE.Color(0x9fcdd2), tmp = new THREE.Color();
    for (let t = 0; t < tc; t++) {
      const a = 3 * t; const up = M.clamp(nrm.getY(a), 0, 1);
      tmp.copy(cBlue).lerp(cSnow, Math.pow(up, 0.5));
      for (let k = 0; k < 3; k++) { col[(a + k) * 3] = tmp.r; col[(a + k) * 3 + 1] = tmp.g; col[(a + k) * 3 + 2] = tmp.b; }
    }
    geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
    const mat = new THREE.MeshStandardMaterial({ vertexColors: true, flatShading: true, roughness: 0.7, metalness: 0.0 });
    this.scene.add(new THREE.Mesh(geo, mat));
  }

  // height of the calving-front top edge as a function of x — big natural variation
  wallTopAt(x) {
    let h = 18;
    h += (this._fbm(x * 0.042 + 5, 1.3) - 0.5) * 26;          // broad towers & saddles
    h += Math.sin(x * 0.13 + this._fbm(x * 0.03, 3) * 4) * 4; // medium bumps
    const notch = this._fbm(x * 0.085 + 12, 2.2);
    if (notch > 0.64) h -= (notch - 0.64) * 46;               // sharp notches / bays
    return Math.max(8, h);
  }

  // z of the wall surface at a given (x, height) — chunks sit flush on this leaning face
  wallFaceZ(x, y) {
    const m = this.glacierMeta;
    const top = this.wallTopAt(x);
    const frac = Math.min(1, Math.max(0, y / top));
    const tw = Math.pow(frac, 1 / 0.74);
    return m.wallFootZ + (m.zLip - m.wallFootZ) * tw;
  }

  buildGlacier() {
    const THREE = this.THREE, M = THREE.MathUtils;
    const cols = 200, rows = 150, stride = cols + 1, verts = [];
    const valleyHalf = 66, wallFootZ = -34, zLip = -46, zBack = -500, wallRows = 30;
    this.glacierMeta = { valleyHalf, wallFootZ, zLip };
    // random large vertical cracks running down the wall (mix of dark recesses + teal-glowing fissures)
    let seed = 7;
    const rnd = () => { seed = (seed * 9301 + 49297) % 233280; return seed / 233280; };
    const cracks = [];
    const nCracks = 9;
    for (let k = 0; k < nCracks; k++) {
      cracks.push({
        x: (rnd() - 0.5) * 2 * valleyHalf * 0.92,
        w: 2.4 + rnd() * 5.5,            // half-width
        depth: 5 + rnd() * 8,            // how deep it cuts into the wall
        glow: rnd() < 0.42,             // some fissures glow teal, the rest are dark
        wander: 1.5 + rnd() * 5,        // sideways meander down the wall
        wfreq: 0.04 + rnd() * 0.13,
        phase: rnd() * 6.28
      });
    }
    this.cracks = cracks;
    const crackAt = (x, y) => {
      let g = 0, glow = 0, dark = 0, depth = 0;
      for (const c of cracks) {
        const cx = c.x + Math.sin(y * c.wfreq + c.phase) * c.wander;
        const dd = Math.abs(x - cx) / c.w;
        if (dd < 1) { const f = Math.pow(1 - dd, 1.7); if (f > g) { g = f; depth = c.depth; } if (c.glow) glow = Math.max(glow, f); else dark = Math.max(dark, f); }
      }
      return { g, glow, dark, depth };
    };
    for (let i = 0; i <= rows; i++) {
      let z, region, tw = 0, tb = 0;
      if (i <= wallRows) {                              // FRONT calving face
        tw = i / wallRows;                              // 0 waterline foot .. 1 lip
        z = M.lerp(wallFootZ, zLip, tw);                // face leans back slightly as it rises
        region = 0;
      } else {                                          // ice ramping back into the land
        tb = (i - wallRows) / (rows - wallRows);        // 0 lip .. 1 far inland
        z = M.lerp(zLip, zBack, tb);
        region = 1;
      }
      const zr = -z;
      for (let j = 0; j <= cols; j++) {
        const u = j / cols;
        const x = (u - 0.5) * 2 * valleyHalf;
        const edge = Math.abs(u - 0.5) * 2;
        const top = this.wallTopAt(x);
        let y, zz = z;
        if (region === 0) {
          y = top * Math.pow(tw, 0.74);                                  // rise from water to the lip
          // mixed surface: some areas rough, some smooth
          const roughMask = M.smoothstep(this._fbm(x * 0.055 + 20, zr * 0.05), 0.42, 0.72);
          y += (this._fbm(x * 0.13 + 4, zr * 0.11) - 0.5) * (1.1 + roughMask * 4.2) * tw;
          zz += (this._fbm(x * 0.05 + 2, zr * 0.05) - 0.5) * 7 * Math.pow(tw, 0.6); // gentle bays & swells
          const cr = crackAt(x, y);                                       // recess the vertical cracks into the wall
          zz -= cr.g * cr.depth * (0.35 + 0.65 * tw);
          if (tw < 0.06) y = M.lerp(0.5, y, tw / 0.06);                  // foot meets the waterline
        } else {
          const climb = Math.pow(tb, 0.95) * (40 + this._fbm(x * 0.05, 9) * 24);
          y = top + climb - tb * 2;                                      // ice ramps gently up toward the horizon
          y += (this._fbm(x * 0.09 + 9, zr * 0.05) - 0.5) * (4 + tb * 12); // gentle inland swells
          y += edge * edge * 8 * tb;                                     // banks up against the valley walls
        }
        verts.push(x, y, zz);
      }
    }
    const idx = [];
    for (let i = 0; i < rows; i++) for (let j = 0; j < cols; j++) {
      const a = i * stride + j, b = a + 1, c = a + stride, d = c + 1;
      idx.push(a, c, b, b, c, d);
    }
    let geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
    geo.setIndex(idx);
    geo = geo.toNonIndexed();
    geo.computeVertexNormals();
    const p = geo.attributes.position, nrm = geo.attributes.normal, tc = p.count / 3;
    const col = new Float32Array(p.count * 3);
    const cIce = new THREE.Color(0xdfeef1), cMid = new THREE.Color(0x9fcfd2), cDeep = new THREE.Color(0x4e9aa0), cShadow = new THREE.Color(0x0d2c33), cGlow = new THREE.Color(0x57e3d2), cTeal = new THREE.Color(0x2c7570);
    const tmp = new THREE.Color();
    for (let t = 0; t < tc; t++) {
      const a = 3 * t;
      const ny = nrm.getY(a), nz = nrm.getZ(a);
      const yc = (p.getY(a) + p.getY(a + 1) + p.getY(a + 2)) / 3;
      const xc = (p.getX(a) + p.getX(a + 1) + p.getX(a + 2)) / 3;
      const up = M.clamp(ny, 0, 1);
      tmp.copy(cDeep).lerp(cMid, Math.pow(up, 0.6)).lerp(cIce, up * up * 0.9);
      if (nz > 0.22) tmp.lerp(cIce, (nz - 0.22) * 0.7);          // camera-facing faces catch light
      const sideways = (1 - up);
      if (sideways > 0.5) tmp.lerp(cShadow, (sideways - 0.5) * 0.9); // crevasse walls in shadow
      const cr = crackAt(xc, yc);
      if (cr.dark > 0.15) tmp.lerp(cShadow, (cr.dark - 0.15) * 1.15); // dark fissures
      if (cr.glow > 0.2) tmp.lerp(cGlow, (cr.glow - 0.2) * 0.66);     // teal-glowing fissures
      if (yc < 6) tmp.lerp(cTeal, M.clamp((6 - yc) / 9, 0, 0.55)); // teal underglow at the waterline
      for (let k = 0; k < 3; k++) { col[(a + k) * 3] = tmp.r; col[(a + k) * 3 + 1] = tmp.g; col[(a + k) * 3 + 2] = tmp.b; }
    }
    geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
    const mat = new THREE.MeshStandardMaterial({ vertexColors: true, flatShading: true, roughness: 0.48, metalness: 0.0, emissive: 0x123f3b, emissiveIntensity: 0.13, side: THREE.DoubleSide });
    const g = new THREE.Mesh(geo, mat);
    this.scene.add(g); this.glacier = g;
  }

  buildSideMountains() {
    const THREE = this.THREE, M = THREE.MathUtils;
    const build = (side) => {
      const rows = 64, cols = 34, stride = cols + 1, verts = [];
      for (let i = 0; i <= rows; i++) {
        const t = i / rows;                          // 0 near shore, 1 far inland
        const z = M.lerp(28, -300, t);
        const valleyEdge = M.lerp(54, 78, Math.pow(t, 0.5)); // foot just outside the ice, framing the valley
        const span = M.lerp(90, 150, t);
        for (let j = 0; j <= cols; j++) {
          const u = j / cols;                        // 0 inner foot (at the ice), 1 outer peak
          let h = Math.pow(u, 0.92) * 116;           // steep rocky walls towering above the ice
          h += this._fbm(u * 2.4 + side * 9, t * 4.2) * 46;
          h += this._fbm(u * 6 + side, t * 9) * 16;
          h *= (0.3 + 0.7 * Math.pow(u, 0.4));
          h += t * 26;                                // walls rise toward the inland saddle
          const x = side * (valleyEdge + u * span) + (this._vnoise(u * 4 + side, t * 6) - 0.5) * 12;
          verts.push(x, h - 4, z);
        }
      }
      const idx = [];
      for (let i = 0; i < rows; i++) for (let j = 0; j < cols; j++) {
        const a = i * stride + j, b = a + 1, c = a + stride, d = c + 1;
        idx.push(a, c, b, b, c, d);
      }
      let geo = new THREE.BufferGeometry();
      geo.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
      geo.setIndex(idx);
      geo = geo.toNonIndexed();
      geo.computeVertexNormals();
      const p = geo.attributes.position, nrm = geo.attributes.normal, tc = p.count / 3;
      const col = new Float32Array(p.count * 3);
      const cRock = new THREE.Color(0x0c151b), cRock2 = new THREE.Color(0x1c2e38), cSnow = new THREE.Color(0x9fb4bb), cBase = new THREE.Color(0x070f14);
      const tmp = new THREE.Color();
      for (let t2 = 0; t2 < tc; t2++) {
        const a = 3 * t2;
        const ny = nrm.getY(a), yc = (p.getY(a) + p.getY(a + 1) + p.getY(a + 2)) / 3;
        const up = M.clamp(ny, 0, 1);
        tmp.copy(cRock).lerp(cRock2, up * 0.6);
        const snow = M.smoothstep(yc, 64, 104) * M.smoothstep(up, 0.66, 0.98); // sparse snow only on high flat ground
        tmp.lerp(cSnow, snow * 0.55);
        if (yc < 5) tmp.lerp(cBase, M.clamp((5 - yc) / 9, 0, 0.8));
        for (let k = 0; k < 3; k++) { col[(a + k) * 3] = tmp.r; col[(a + k) * 3 + 1] = tmp.g; col[(a + k) * 3 + 2] = tmp.b; }
      }
      geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
      const mat = new THREE.MeshStandardMaterial({ vertexColors: true, flatShading: true, roughness: 1.0, metalness: 0.0, side: THREE.DoubleSide });
      this.scene.add(new THREE.Mesh(geo, mat));
    };
    build(-1); build(1);
  }

  makeIceChunk(w, h, d) {
    const THREE = this.THREE;
    const geo = new THREE.BoxGeometry(w, h, d, 3, 3, 3);
    const p = geo.attributes.position;
    for (let i = 0; i < p.count; i++) {
      const x = p.getX(i), y = p.getY(i), z = p.getZ(i);
      p.setX(i, x + (this._vnoise(x * 1.5 + i, y * 1.5) - 0.5) * w * 0.16);
      p.setY(i, y + (this._vnoise(y * 1.5 + i, z * 1.5) - 0.5) * h * 0.2);
      p.setZ(i, z + (this._vnoise(z * 1.5 + i, x * 1.5) - 0.5) * d * 0.2);
    }
    const ng = geo.toNonIndexed();
    ng.computeVertexNormals();
    const cc = ng.attributes.position.count;
    const col = new Float32Array(cc * 3);
    // match the wall ice palette so a calved chunk reads as part of the wall
    const cIce = new THREE.Color(0xdfeef1), cMid = new THREE.Color(0x9fcfd2), cDeep = new THREE.Color(0x4e9aa0), cShadow = new THREE.Color(0x214b52);
    const nrm = ng.attributes.normal, tmp = new THREE.Color();
    for (let t = 0; t < cc / 3; t++) {
      const a = 3 * t, ny = nrm.getY(a), nz = nrm.getZ(a);
      const up = Math.max(0, ny);
      tmp.copy(cDeep).lerp(cMid, Math.pow(up, 0.6)).lerp(cIce, Math.max(0, ny * 0.5 + nz * 0.5));
      if (ny < -0.1) tmp.lerp(cShadow, 0.5);
      for (let k = 0; k < 3; k++) { col[(a + k) * 3] = tmp.r; col[(a + k) * 3 + 1] = tmp.g; col[(a + k) * 3 + 2] = tmp.b; }
    }
    ng.setAttribute('color', new THREE.BufferAttribute(col, 3));
    const mat = new THREE.MeshStandardMaterial({ vertexColors: true, flatShading: true, roughness: 0.46, metalness: 0.0, emissive: 0x123f3b, emissiveIntensity: 0.06 });
    return new THREE.Mesh(ng, mat);
  }

  setupInteraction() {
    const THREE = this.THREE;
    this.raycaster = new THREE.Raycaster();
    this.pointer = new THREE.Vector2();
    const dom = this.renderer.domElement;
    this._onMove = (e) => {
      const r = dom.getBoundingClientRect();
      this.pointer.x = ((e.clientX - r.left) / r.width) * 2 - 1;
      this.pointer.y = -((e.clientY - r.top) / r.height) * 2 + 1;
      this.checkHover();
    };
    dom.addEventListener('pointermove', this._onMove);
    dom.addEventListener('pointerdown', this._onMove);
  }

  checkHover() {
    if (!this.raycaster || !this.blocks) return;
    this.raycaster.setFromCamera(this.pointer, this.camera);
    const meshes = this.blocks.filter(b => !b.userData.calving && !b.userData.gone).map(b => b.userData.hit);
    const hits = this.raycaster.intersectObjects(meshes, false);
    if (hits.length) this.calve(hits[0].object.userData.block);
  }

  calve(blk) {
    const u = blk.userData;
    if (u.calving || u.gone) return;
    u.calving = true;
    const styles = ['topple', 'drop', 'slump'];
    u.style = styles[(Math.random() * 3) | 0];
    if (u.style === 'topple') {            // tips forward off the wall and crashes into the sea
      u.vy = 0.6; u.vz = 1.6 + Math.random() * 0.8; u.vx = (Math.random() - 0.5) * 0.5;
      u.angX = 1.9 + Math.random() * 0.7; u.angZ = (Math.random() - 0.5) * 0.5;
    } else if (u.style === 'drop') {       // breaks loose and falls more or less straight down
      u.vy = -0.6; u.vz = 0.3 + Math.random() * 0.4; u.vx = (Math.random() - 0.5) * 0.3;
      u.angX = (Math.random() - 0.2) * 0.7; u.angZ = (Math.random() - 0.5) * 0.5;
    } else {                               // slumps and sinks into the water in place
      u.vy = -1.1; u.vz = 0.4; u.vx = 0; u.compress = 0;
    }
  }

  makeSplash(x, z, size) {
    const THREE = this.THREE;
    if (!this._splashTex) this._splashTex = this.makeSplashTexture();
    const m = new THREE.Mesh(
      new THREE.CircleGeometry(1, 28),
      new THREE.MeshBasicMaterial({ map: this._splashTex, transparent: true, opacity: 0.85, depthWrite: false })
    );
    m.rotation.x = -Math.PI / 2; m.position.set(x, 0.42, z);
    this.scene.add(m);
    this.splashes.push({ mesh: m, life: 0, size: size });
  }

  shatter(blk, scl) {
    const THREE = this.THREE;
    scl = scl || 1;
    const n = Math.round((5 + Math.floor(Math.random() * 3)) * scl);
    for (let i = 0; i < n; i++) {
      const s = (0.8 + Math.random() * 1.5) * (0.7 + scl * 0.4);
      const m = this.makeIceChunk(s, s * 0.6, s * 0.8);
      m.position.set(blk.position.x + (Math.random() - 0.5) * 4, Math.max(0.5, blk.position.y), blk.position.z + (Math.random() - 0.5) * 3);
      const ang = Math.random() * Math.PI * 2, sp = (1.5 + Math.random() * 4) * scl;
      m.userData = {
        vx: Math.cos(ang) * sp, vz: Math.sin(ang) * sp + 1.2, vy: (1 + Math.random() * 3) * scl,
        vr: new THREE.Vector3((Math.random() - 0.5) * 4, (Math.random() - 0.5) * 4, (Math.random() - 0.5) * 4),
        rest: 0.3 + Math.random() * 0.4, dx: Math.cos(ang) * (0.4 + Math.random() * 0.6), dz: Math.sin(ang) * 0.4 + 0.25,
        spin: (Math.random() - 0.5) * 0.6, bob: Math.random() * 6.28, floating: false
      };
      this.scene.add(m);
      this.fragments.push(m);
    }
    while (this.fragments.length > 70) { const old = this.fragments.shift(); this.scene.remove(old); old.geometry.dispose(); }
  }

  stepBlocks(dt, t) {
    if (!this.blocks) return;
    const g = 26, waterY = 0.7;
    for (const blk of this.blocks) {
      const u = blk.userData;
      if (u.gone || !u.calving) continue;          // un-hovered chunks stay part of the wall (static)
      // reveal the scar cavity + engraved description as the chunk leaves
      u.reveal = Math.min(1, u.reveal + dt * 1.7);
      u.scar.material.opacity = u.reveal * 0.95;
      if (u.descEl) u.descEl.style.opacity = Math.max(0, (u.reveal - 0.3) / 0.7).toFixed(3);
      if (u.style === 'slump') {
        u.compress = Math.min(1, u.compress + dt * 0.85);
        blk.scale.y = 1 - u.compress * 0.82;
        u.vy -= g * 0.22 * dt;
        blk.position.y += u.vy * dt;
        blk.position.z += u.vz * dt;
        blk.rotation.z += 0.22 * dt;
        if (blk.position.y <= waterY || u.compress >= 1) {
          this.makeSplash(blk.position.x, blk.position.z, 0.6);
          this.shatter(blk, 0.55);
          u.gone = true; blk.visible = false;
        }
      } else {
        u.vy -= g * dt;
        blk.position.x += u.vx * dt;
        blk.position.y += u.vy * dt;
        blk.position.z += u.vz * dt;
        if (u.style === 'topple') u.angX += 1.5 * dt;   // forward pitch accelerates
        blk.rotation.x -= u.angX * dt;
        blk.rotation.z += u.angZ * dt;
        if (blk.position.y <= waterY) {
          this.makeSplash(blk.position.x, blk.position.z, 1.0);
          this.shatter(blk, 1.0);
          u.gone = true; blk.visible = false;
        }
      }
    }
    if (this.fragments) {
      for (const f of this.fragments) {
        const u = f.userData;
        if (!u.floating) {
          u.vy -= g * dt;
          f.position.x += u.vx * dt; f.position.y += u.vy * dt; f.position.z += u.vz * dt;
          f.rotation.x += u.vr.x * dt; f.rotation.y += u.vr.y * dt; f.rotation.z += u.vr.z * dt;
          if (f.position.y <= u.rest) { f.position.y = u.rest; u.floating = true; }
        } else {
          f.position.x += u.dx * dt; f.position.z += u.dz * dt;
          f.position.y = u.rest + Math.sin(t * 1.3 + u.bob) * 0.15;
          f.rotation.z += u.spin * dt;
          u.dx *= 0.992; u.dz *= 0.992;
        }
      }
    }
    if (this.splashes) {
      for (let i = this.splashes.length - 1; i >= 0; i--) {
        const s = this.splashes[i]; s.life += dt;
        const k = s.life / 0.9;
        s.mesh.scale.setScalar((0.5 + k * 3.2) * s.size);
        s.mesh.material.opacity = Math.max(0, 0.75 * (1 - k));
        if (k >= 1) { this.scene.remove(s.mesh); s.mesh.geometry.dispose(); this.splashes.splice(i, 1); }
      }
    }
  }
}
